    function validateConference() {
      let allMatch = true;
      let feedbackErrors = [];

      currentNota.items.forEach((it, idx) => {
        const input = document.getElementById(`countInput_${idx}`);
        const statusCol = document.getElementById(`statusCol_${idx}`);
        const counted = parseFloat(input.value) || 0;

        it.qtdContada = counted; // armazena contagem

        if (counted === it.qtdNf) {
          input.classList.remove("has-error");
          statusCol.innerHTML = `<span class="tag-match tag-ok">✓ Bateu</span>`;
        } else {
          allMatch = false;
          input.classList.add("has-error");
          statusCol.innerHTML = `<span class="tag-match tag-alert">✕ Divergente</span>`;
          feedbackErrors.push(`Item ${it.codigo}: Contado ${counted} ${it.un} | Esperado no Pedido: ${it.qtdPedido} ${it.un}`);
        }
      });

      const feedbackDiv = document.getElementById("conferenceFeedback");

      if (allMatch) {
        // Sucesso 100%!
        feedbackDiv.innerHTML = `
          <div class="status-toast" style="background: #dcfce7; color: #15803d; border: 1px solid #86efac; margin-top: 14px;">
            <div>
              <b>🎉 CONFERÊNCIA CONCLUÍDA COM SUCESSO!</b><br>
              • Quantidades 100% validadas com o Pedido Sienge (${currentNota.pedidoSienge}).<br>
              • Documento de Entrada e Estoque criados no Sienge.<br>
              • Título no Contas a Pagar gerado com sucesso.
            </div>
          </div>

    function executeSupervisorRelease() {
      const email = document.getElementById("supEmail").value.trim();
      const pass = document.getElementById("supPassword").value.trim();
      const autoAdjust = document.getElementById("chkAutoAdjust").checked;

      if (!email || !pass) {
        alert("Informe o e-mail e a senha do supervisor para autenticar a liberação.");
        return;
      }

      const notaLiberada = pendingNotas[selectedPendingIndex];

      alert(`✅ SUCESSO!\n\nNota NF ${notaLiberada.numeroNf} liberada pelo supervisor (${email}).\n${autoAdjust ? "• Chamada API Sienge: Pedido " + notaLiberada.pedidoSienge + " ajustado automaticamente para refletir a quantidade contada.\n" : ""}• Recebimento integrado e Contas a Pagar gerado.`);

      // Remove dos pendentes
      pendingNotas.splice(selectedPendingIndex, 1);
      updatePendingCount();
      closeModal("modalPending");
    }
